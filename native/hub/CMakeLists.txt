cmake_minimum_required(VERSION 3.21)
project(REgenOSHub VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(hub-ui STATIC hub.cpp hub.h)
target_link_libraries(hub-ui PUBLIC Qt6::Widgets Qt6::Network)
target_include_directories(hub-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(hub-ui PRIVATE /W4)
else()
  target_compile_options(hub-ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(regenos-hub main.cpp)
target_link_libraries(regenos-hub PRIVATE hub-ui)
install(TARGETS regenos-hub RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(hub-tests tests.cpp)
  target_link_libraries(hub-tests PRIVATE hub-ui Qt6::Test)
  add_test(NAME hub-widgets COMMAND hub-tests)
  set_tests_properties(hub-widgets PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
